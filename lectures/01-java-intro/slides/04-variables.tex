\section{Core concepts}

\begin{frame}{Variables and types}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \lead{Declaration.}
      \begin{semiverbatim}
int count = 42;
double pi = 3.14;
boolean ok = true;
String name = "Java";
      \end{semiverbatim}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Primitive types}
        \texttt{byte}, \texttt{short}, \texttt{int}, \texttt{long},
        \texttt{float}, \texttt{double}, \texttt{char}, \texttt{boolean}
      \end{block}
      \medskip
      Reference types (e.g.\ \texttt{String}) live on the heap;
      primitives hold values directly.
    \end{column}
  \end{columns}
\end{frame}
