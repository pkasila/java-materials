\section{Getting started}

\begin{frame}{Hello, World!}
  \lead{Java} is a statically typed, object-oriented language that runs on the JVM.

  \medskip
  Every program starts with a \texttt{main} method:

  \smallskip
  \begin{semiverbatim}
public class Hello {
  public static void main(String[] args) {
    System.out.println("Hello, World!");
  }
}
  \end{semiverbatim}
\end{frame}
