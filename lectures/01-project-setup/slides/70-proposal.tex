\begin{frame}{Первичный Project Proposal}
  \lead{Короткий документ в репозитории} (раздел README или \texttt{PROPOSAL.md}).

  \medskip
  \begin{enumerate}\setlength\itemsep{0.32em}
    \item \textbf{Тема:} номер и название из списка \textbf{или} согласованная своя.
    \item \textbf{Проблема:} 1--2 абзаца, что ломается сегодня без системы.
    \item \textbf{Аудитория:} кто пользователь, в каком контексте.
    \item \textbf{Сущности:} ориентир 5--7 взаимосвязанных; можно опереться на таблицу.
    \item \textbf{Путь:} базовый / стандартный / продвинутый (базовый --- по умолчанию).
    \item \textbf{Scope v1:} что входит и что сознательно откладывается.
  \end{enumerate}
\end{frame}
