\begin{frame}{Git с первого дня}
  \lead{История и remote --- часть проекта, не \flqq потом зальём\frqq.}

  \medskip
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Локально}
        \begin{itemize}
          \item \texttt{git init} (или New Project в IDEA);
          \item осмысленные коммиты;
          \item не коммитить сборку и кэш IDE.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Remote проекта}
        \begin{itemize}
          \item GitHub \textbf{или} GitLab --- на ваш выбор;
          \item приватный репозиторий допустим;
          \item это не контур сдачи лабораторных.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
