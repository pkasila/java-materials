\begin{frame}{Зона ответственности оргкомитета III этапа}
  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{Обеспечивает}
        \begin{itemize}\setlength\itemsep{0.28em}
          \item определение площадки проведения;
          \item предоставление и обслуживание рабочих мест участников;
          \item канал связи и \textbf{ограничение доступа к сети Интернет}
                за пределами АТС;
          \item получение от Оргкомитета сведений для доступа к АТС
                и подключение рабочих мест;
          \item порядок на площадке и сопровождение участников
                уполномоченными представителями.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Не делает}
        \begin{itemize}\setlength\itemsep{0.28em}
          \item не разрабатывает олимпиадные задания;
          \item не формирует жюри;
          \item не устанавливает и не предлагает квоты на IV этап;
          \item не проверяет и не оценивает работы.
        \end{itemize}
      \end{block}
      \begin{block}{Республиканский Оргкомитет}
        Разработка заданий и наборов данных, АТС и её работоспособность, апелляции.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
