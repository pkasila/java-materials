\begin{frame}{Правила проведения туров на площадке}
  \vspace{0.2em}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \lead{Режим участника}
      \begin{itemize}\setlength\itemsep{0.3em}
        \item нахождение на закреплённом рабочем месте весь тур;
        \item кратковременные выходы --- только в сопровождении;
        \item любое общение между участниками запрещено;
        \item вопросы --- исключительно через АТС и (или)
              уполномоченному представителю;
        \item решение должно быть зафиксировано в АТС.
      \end{itemize}
    \end{column}
    \begin{column}{0.49\textwidth}
      \lead{Запреты}
      \begin{itemize}\setlength\itemsep{0.3em}
        \item мобильные телефоны, средства связи, иные неразрешённые
              технические средства;
        \item внешние генеративные ИИ-сервисы и ассистенты кода;
        \item заранее подготовленный или полученный от третьих лиц код,
              передача своих решений другим;
        \item личные заметки, бумажные и электронные носители.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
