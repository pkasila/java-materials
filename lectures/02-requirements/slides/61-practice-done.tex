\begin{frame}{Критерий готовности}
  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{Готово, если}
        \begin{itemize}
          \item в клоне есть Markdown с Proposal и backlog;
          \item истории про \emph{вашу} проблему, не про абстрактный CRUD;
          \item MVP короче полного списка: есть Won't / \flqq не в v1\frqq;
          \item есть диаграмма акторов; у историй есть AC.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Ещё не готово}
        \begin{itemize}
          \item backlog только в чате или в голове;
          \item одна строка \flqq CRUD всех сущностей\frqq;
          \item 50 пунктов без приоритета и без MVP;
          \item диаграмма классов вместо акторов.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
