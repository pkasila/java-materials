\begin{frame}{NFR для студенческого MVP}
  \lead{Конкретно и проверяемо.} Не SLA и не \flqq высокая нагрузка\frqq.

  \medskip
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{Подходит}
        \begin{itemize}\setlength\itemsep{0.28em}
          \item пароль не пишется в лог;
          \item ошибка валидации видна пользователю;
          \item сборка --- одна команда \texttt{./gradlew build};
          \item пустое название задачи нельзя сохранить.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Не подходит в v1}
        \begin{itemize}\setlength\itemsep{0.28em}
          \item \flqq система должна быть удобной\frqq;
          \item uptime 99,9\,\%;
          \item миллион пользователей;
          \item \flqq промышленный уровень безопасности\frqq.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
