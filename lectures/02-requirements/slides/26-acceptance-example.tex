\begin{frame}{AC к истории про владельца}
  \vspace{0.15em}
  {\small Как участник, я хочу назначить владельца задачи \ldots}

  \vspace{0.45em}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{Чеклист}
        \begin{itemize}\setlength\itemsep{0.22em}
          \item у задачи ровно один владелец;
          \item владельца можно сменить;
          \item назначенный владелец отображается в карточке задачи.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Given / When / Then}
        {\small
        \textbf{Дано} участник и задача без владельца.\\
        \textbf{Когда} участник назначает владельца.\\
        \textbf{Тогда} владелец сохраняется
        и отображается в карточке задачи.}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
