\begin{frame}{AC к истории про владельца}

  {\small Как участник, я хочу назначить владельца задачи \ldots}

  \vspace{0.45em}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{Чеклист}
        \begin{itemize}\setlength\itemsep{0.22em}
          \item у задачи ровно один владелец;
          \item владельца можно сменить;
          \item без владельца нельзя перевести в \flqq в работе\frqq.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Given / When / Then}
        {\small
        \textbf{Дано} задача без владельца.\\
        \textbf{Когда} статус меняют на \flqq в работе\frqq.\\
        \textbf{Тогда} система отказывает
        и просит назначить владельца.}
      \end{block}
    \end{column}
  \end{columns}
